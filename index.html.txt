<script>
    // यहाँ आप अपने सवाल और जवाब बदल सकते हैं
    const quizData = [
        {
            question: "रासायनिक अभिक्रिया में भाग लेने वाले पदार्थ क्या कहलाते हैं?",
            options: ["उत्पाद (Product)", "अभिकारक (Reactant)", "उत्प्रेरक (Catalyst)", "यौगिक (Compound)"],
            answer: 1
        },
        {
            question: "प्रकाश के परावर्तन के कितने नियम हैं?",
            options: ["एक", "दो", "तीन", "चार"],
            answer: 1
        },
        {
            question: "जल का रासायनिक सूत्र क्या है?",
            options: ["CO2", "H2O", "O2", "NaCl"],
            answer: 1
        },
        {
            question: "हिंदी वर्णमाला में कुल कितने स्वर हैं?",
            options: ["9", "11", "13", "33"],
            answer: 1
        },
        {
            question: "भारत की राजधानी क्या है?",
            options: ["मुंबई", "कोलकाता", "नई दिल्ली", "चेन्नई"],
            answer: 2
        }
    ];

    const TIME_PER_QUESTION = 15; // सेकंड्स
    let currentQuestionIndex = 0;
    let score = 0;
    let timer;
    let timeLeft = TIME_PER_QUESTION;
    let selectedAnswer = null;
    let isAnswered = false;

    const questionMeta = document.getElementById('question-meta');
    const questionText = document.getElementById('question-text');
    const optionsContainer = document.getElementById('options-container');
    const nextBtn = document.getElementById('next-btn');
    const quizScreen = document.getElementById('quiz-screen');
    const resultScreen = document.getElementById('result-screen');
    const leaderboardScreen = document.getElementById('leaderboard-screen');

    function startQuiz() {
        currentQuestionIndex = 0;
        score = 0;
        loadQuestion();
    }

    function loadQuestion() {
        resetState();
        const currentQ = quizData[currentQuestionIndex];
        questionMeta.innerText = `प्रश्न ${currentQuestionIndex + 1}/${quizData.length}`;
        questionText.innerText = currentQ.question;

        currentQ.options.forEach((option, index) => {
            const button = document.createElement('button');
            button.innerText = option;
            button.classList.add('option-btn');
            button.onclick = () => selectOption(button, index);
            optionsContainer.appendChild(button);
        });

        startTimer();
    }

    function resetState() {
        clearInterval(timer);
        timeLeft = TIME_PER_QUESTION;
        isAnswered = false;
        selectedAnswer = null;
        document.getElementById('timer').innerText = `समय: ${timeLeft}s`;
        nextBtn.disabled = true;
        while (optionsContainer.firstChild) {
            optionsContainer.removeChild(optionsContainer.firstChild);
        }
    }

    function startTimer() {
        timer = setInterval(() => {
            timeLeft--;
            document.getElementById('timer').innerText = `समय: ${timeLeft}s`;
            if (timeLeft <= 0) {
                clearInterval(timer);
                autoSelectWrong();
            }
        }, 1000);
    }

    function selectOption(button, index) {
        if (isAnswered) return;
        isAnswered = true;
        clearInterval(timer);
        selectedAnswer = index;

        const correct = quizData[currentQuestionIndex].answer;
        const buttons = optionsContainer.getElementsByClassName('option-btn');

        if (index === correct) {
            button.classList.add('correct');
            score++;
        } else {
            button.classList.add('wrong');
            buttons[correct].classList.add('correct');
        }

        Array.from(buttons).forEach(btn => btn.style.pointerEvents = 'none');
        nextBtn.disabled = false;
    }

    function autoSelectWrong() {
        isAnswered = true;
        const buttons = optionsContainer.getElementsByClassName('option-btn');
        Array.from(buttons).forEach(btn => btn.style.pointerEvents = 'none');
        const correct = quizData[currentQuestionIndex].answer;
        buttons[correct].classList.add('correct');
        nextBtn.disabled = false;
    }

    function nextQuestion() {
        currentQuestionIndex++;
        if (currentQuestionIndex < quizData.length) {
            loadQuestion();
        } else {
            showResult();
        }
    }

    function showResult() {
        quizScreen.style.display = 'none';
        resultScreen.style.display = 'block';
        document.getElementById('score').innerText = score;
        document.getElementById('total-score').innerText = quizData.length;
    }

    function saveScore() {
        const nameInput = document.getElementById('student-name');
        const name = nameInput.value.trim();
        if (!name) {
            alert("कृपया अपना नाम दर्ज करें!");
            return;
        }

        let leadData = JSON.parse(localStorage.getItem('gyanalayLeaderboard')) || [];
        leadData.push({ name: name, score: score, date: new Date().toLocaleDateString() });
        leadData.sort((a, b) => b.score - a.score);
        localStorage.setItem('gyanalayLeaderboard', JSON.stringify(leadData));

        resultScreen.style.display = 'none';
        showLeaderboard();
    }

    function showLeaderboard() {
        leaderboardScreen.style.display = 'block';
        const tbody = document.getElementById('leaderboard-body');
        tbody.innerHTML = '';

        let leadData = JSON.parse(localStorage.getItem('gyanalayLeaderboard')) || [];
        
        leadData.slice(0, 10).forEach((entry, index) => {
            const row = `<tr>
                <td>#${index + 1}</td>
                <td>${entry.name}</td>
                <td>${entry.score}</td>
            </tr>`;
            tbody.innerHTML += row;
        });
    }

    function restartQuiz() {
        leaderboardScreen.style.display = 'none';
        quizScreen.style.display = 'block';
        startQuiz();
    }

    startQuiz();
</script>
